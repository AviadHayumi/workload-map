func scaleFixture(b *testing.B, intermediate bool) ([]*corev1.Pod, []*metav1.PartialObjectMetadata) {
	b.Helper()
	count := 1000
	if raw := os.Getenv("OWNER_BENCH_ROOTS"); raw != "" {
		n, err := strconv.Atoi(raw)
		if err != nil || n < 1 || n > 100000 {
			b.Fatal("invalid OWNER_BENCH_ROOTS")
		}
		count = n
	}
	pods := make([]*corev1.Pod, 0, count)
	objects := make([]*metav1.PartialObjectMetadata, 0, count*3)
	controller := true
	for i := range count {
		root := &metav1.PartialObjectMetadata{TypeMeta: metav1.TypeMeta{APIVersion: "batch/v1", Kind: "Job"}, ObjectMeta: metav1.ObjectMeta{Name: fmt.Sprintf("root-%d", i), Namespace: "scale", UID: types.UID(fmt.Sprintf("root-%d", i)), ResourceVersion: "1"}}
		objects = append(objects, root)
		owner := metav1.OwnerReference{APIVersion: "batch/v1", Kind: "Job", Name: root.Name, UID: root.UID, Controller: &controller}
		if intermediate {
			middle := &metav1.PartialObjectMetadata{TypeMeta: metav1.TypeMeta{APIVersion: "apps/v1", Kind: "ReplicaSet"}, ObjectMeta: metav1.ObjectMeta{Name: fmt.Sprintf("middle-%d", i), Namespace: "scale", UID: types.UID(fmt.Sprintf("middle-%d", i)), ResourceVersion: "1", OwnerReferences: []metav1.OwnerReference{owner}}}
			objects = append(objects, middle)
			owner = metav1.OwnerReference{APIVersion: "apps/v1", Kind: "ReplicaSet", Name: middle.Name, UID: middle.UID, Controller: &controller}
		}
		pod := &corev1.Pod{TypeMeta: metav1.TypeMeta{APIVersion: "v1", Kind: "Pod"}, ObjectMeta: metav1.ObjectMeta{Name: fmt.Sprintf("pod-%d", i), Namespace: "scale", UID: types.UID(fmt.Sprintf("pod-%d", i)), ResourceVersion: "1", OwnerReferences: []metav1.OwnerReference{owner}}}
		pods = append(pods, pod)
		objects = append(objects, &metav1.PartialObjectMetadata{TypeMeta: pod.TypeMeta, ObjectMeta: pod.ObjectMeta})
	}
	return pods, objects
}
